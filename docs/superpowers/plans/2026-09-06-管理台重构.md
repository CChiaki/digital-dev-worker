# 管理台重构 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 管理台路由化重构 + 数字员工档案后台化（能力绑定分派校验）+ 任务 AI 解析创建与 draft 发布态 + 员工下钻与直播节点分组 + 服务端消息中心 + 钉钉/企微/Webhook 通知渠道。

**Architecture:** 后端（packages/console-api）新增四个同模式 file store（employees/messages/notification-channels + 已有 capabilities），消息由 TaskStore/EventStore 装饰器（message-hub）在 append/finish 时生成并推送渠道；分派能力过滤通过 scheduler 的 `canDispatch` 钩子 + `ManagedRoster`（EmployeeStore 支撑）实现，runtime 仅扩展 `acquire` 的可选谓词参数。前端（apps/console）引入 vue-router 重构为侧边栏布局，新增 Dashboard/员工管理/员工详情/通知渠道四页并重构员工直播。

**Tech Stack:** TypeScript monorepo（pnpm）、node:sqlite、vitest、Vue3 + element-plus + vue-router@4。

**Spec:** `docs/superpowers/specs/2026-09-06-管理台重构-design.md`

## Global Constraints

- 现有页面（能力管理/审计台账/人工放行）功能零删减，只换布局挂载方式。
- `GET /api/tasks` 新增查询参数不传时行为不变；**允许增补可选字段**（2026-09-06 Task 8 裁决：列表行补 `reply` 字段属最小一致化，非回归）；`POST /api/tasks` 除「创建默认 draft」外请求格式与校验零变化。
- 创建即 draft 是**有意的行为变化**（发布才可被接取）；现有脚本/e2e 创建任务处补 publish 调用。
- 员工删除 = 停用（enabled=false），不物理删。
- `EmployeeProfile`（runtime）类型不动；能力绑定存 console-api `EmployeeRecord.capabilities`，分派过滤在装配层做。
- **员工模型绑定（一人一模型一 key）**：绑定存 `EmployeeRecord.model`（与 ModelSpec 同构）；专属 gateway 在装配层注入（inproc executor `modelFor` / fork job `employeeRoute`）；未绑定回退全局 routes 零回归；能力注册表与模型彻底解耦（kind 只管工具集）；GET 脱敏 apiKey、PUT 留空/'***' 保原 key。
- 老 yaml `employees:` 段仅在 employees.json 不存在时作 seed，之后以管理台为准。
- 消息只保留最近 1000 条；通知渠道推送失败只记事件日志，不阻断、不重试。
- 提交规范：中文 conventional commit + `Co-Authored-By: Claude Code <noreply@anthropic.com>`。
- 严禁测试监听端口/启动服务（handler 纯函数直接调用）；打包/启动 demo 须用户单独授权。
- 测试命令：`pnpm -C packages/console-api test`、`pnpm -C packages/runtime test`、`pnpm -C apps/console test`。

```ts
    ).run(taskId);
    if (r.changes === 0) {
      const rec = await this.get(taskId);
      throw new Error(rec ? `任务状态为 ${rec.status}，仅 draft 任务可发布` : `任务不存在: ${taskId}`);
```

```ts
import { parseTaskPackage, type ModelGateway } from '@ddw/runtime';

/** 从模型回复中提取 yaml 代码块（```yaml ... ```；无围栏时整段原文兜底） */
export function extractYamlBlock(reply: string): string {
  const fenced = reply.match(/```(?:yaml)?\s*\n([\s\S]*?)```/);
  return (fenced?.[1] ?? reply).trim();
}

/** 系统提示词：任务包 schema 约束 + 仅允许的 kind 白名单（生成 kind 必须取自注册表） */
export function buildTaskParseSystemPrompt(kinds: string[]): string {
  return [
    '你是研发任务包生成器。把用户的自然语言需求描述转换为任务包 YAML（只输出 YAML，不要解释）。',
    '严格遵循如下 schema：',
    'taskId: <kebab-case 唯一标识>, title: <中文标题>,',
    'repo: { url: <git 仓库地址>, branch: <分支>, baseBranch?: <基线分支> },',
    'role?: <岗位>, dependsOn?: [任务包id],',
    'plan: [ { id: <t1/t2...>, kind?: <能力类型>, title: <项标题>, detail: <该项要做什么>, verify?: <验证命令> } ]',
    `kind 只能取以下已注册值（缺省 dev）：${kinds.join(', ') || 'dev'}。`,
    'plan 与 tasks 互斥，用 plan。每个 plan 项的 detail 要具体可执行；关键项（commit/test）建议配 verify 命令。',
    '不要编造仓库地址——若用户未给出 repo.url，用占位 "http://localhost:3000/demo/CHANGE-ME.git" 并在 title 后加「（仓库待定）」。',
  ].join('\n');
}

/** AI 解析：用户描述 → 任务包 yaml（过 parseTaskPackage 与 kind 白名单双重校验，失败抛可读错误） */
export async function parseTaskDescription(
  gateway: ModelGateway, description: string, enabledKinds: () => Promise<string[]>,
): Promise<string> {
  const kinds = await enabledKinds();
  const stream = gateway.streamFnFor('chat');
  const reply = await stream(gateway.modelFor('chat'), {
    systemPrompt: buildTaskParseSystemPrompt(kinds),
    messages: [{ role: 'user', content: description, timestamp: Date.now() }],
  } as never) as string;
  const yaml = extractYamlBlock(reply);
  try {
    parseTaskPackage(yaml);
  } catch (e) {
    throw new Error(`智能生成失败：模型输出不是合法任务包（${e instanceof Error ? e.message : String(e)}）`);
  }
  const pkg = parseTaskPackage(yaml);
  for (const item of pkg.plan ?? []) {
    const kind = item.kind ?? 'dev';
    if (!kinds.includes(kind)) {
      throw new Error(`智能生成失败：模型输出的任务项 ${item.id}.kind='${kind}' 未注册（可用: ${kinds.join(', ')}），请调整描述或改用手动模式`);
    }
  }
  return yaml;
}
```

```ts
/** 数字员工档案（后台化，2026-09-06）：capabilities 绑定能力注册表 kind（空=全部可用） */
export interface EmployeeRecord {
  id: string;
  name: string;
  role: string;
  /** 可接任务岗位集合（EmployeeProfile.skills 语义不变） */
  skills: string[];
  /** 能力 kind 绑定：任务项 kind ∈ 此集合才可执行该项；空/缺省 = 全部可用 */
  capabilities: string[];
  /** 专属模型（缺省 = 回退全局 routes）；apiKey 支持 enc:v1: 密文 */
  model?: EmployeeModelBinding;
  supervision?: { level: 'shadow' | 'assisted' | 'trusted' };
  /** false = 停用（调度不再分派；不物理删） */
  enabled: boolean;
  createdAt: number;
}

const LEVELS = new Set(['shadow', 'assisted', 'trusted']);

export function validateEmployeeRecord(rec: EmployeeRecord, capabilityKinds: string[]): void {
  if (typeof rec.id !== 'string' || !rec.id.trim()) throw new Error('员工 id 必须是非空字符串');
  if (typeof rec.name !== 'string' || !rec.name.trim()) throw new Error('员工 name 必须是非空字符串');
  if (typeof rec.role !== 'string' || !rec.role.trim()) throw new Error('员工 role 必须是非空字符串');
  if (!Array.isArray(rec.skills) || !rec.skills.every((x) => typeof x === 'string')) {
    throw new Error('员工 skills 必须是字符串数组');
  }
  if (!Array.isArray(rec.capabilities) || !rec.capabilities.every((x) => typeof x === 'string')) {
    throw new Error('员工 capabilities 必须是字符串数组');
  }
  const unknown = rec.capabilities.filter((k) => !capabilityKinds.includes(k));
  if (unknown.length > 0) throw new Error(`员工 capabilities 含未注册的能力: ${unknown.join(', ')}（允许: ${capabilityKinds.join(', ') || '无'}）`);
  if (rec.supervision !== undefined && !LEVELS.has(rec.supervision.level)) {
    throw new Error('员工 supervision.level 必须是 shadow | assisted | trusted');
  }
  if (typeof rec.enabled !== 'boolean') throw new Error('员工 enabled 必须是布尔值');
  if (rec.model !== undefined) {
    const m = rec.model;
    if (typeof m.baseUrl !== 'string' || !m.baseUrl.trim()) throw new Error('员工 model.baseUrl 必须是非空字符串');
    if (typeof m.apiKey !== 'string' || !m.apiKey.trim()) throw new Error('员工 model.apiKey 必须是非空字符串');
    if (typeof m.model !== 'string' || !m.model.trim()) throw new Error('员工 model.model 必须是非空字符串');
    if (m.api !== undefined && m.api !== 'openai-completions' && m.api !== 'anthropic-messages') {
      throw new Error('员工 model.api 必须是 openai-completions | anthropic-messages');
    }
  }
}
```

```ts
import type { EmployeeProfile, ModelSpec } from '@ddw/runtime';
import type { EmployeeRecord, EmployeeStore } from './employee-store.js';
import type { TaskRecord } from '../stores/index.js';

/** 调度器名册最小接口（runtime EmployeeRoster 结构兼容；scheduler 依赖此类型便于注入） */
export type SchedulerRoster = {
  acquire(role?: string, extra?: (p: EmployeeProfile) => boolean): EmployeeProfile | null;
  release(id: string): void;
};

const kindsOf = (task: TaskRecord): string[] =>
  (task.pkg.plan ?? []).map((it) => it.kind ?? 'dev');

/**
 * 员工档案支撑的名册（2026-09-06 后台化）：EmployeeStore 为唯一事实源。
 * refresh() 由 pipeline 每 tick 前调用（后台改动下个 tick 生效，无需重启）；
 * canDispatch 是同步谓词（基于 refresh 时的档案快照），供 scheduler 分派前过滤
 * 「plan 任务项 kind ∈ 员工 capabilities（空=全部可用；无 plan 不限）」。
 */
export class ManagedRoster implements SchedulerRoster {
  private free = new Set<string>();
  private records: EmployeeRecord[] = [];

  constructor(private readonly store: EmployeeStore) {}

  async refresh(): Promise<void> {
    this.records = (await this.store.list()).filter((r) => r.enabled);
    this.free = new Set(this.records.map((r) => r.id));
    // 注意：refresh 会重置占用状态——生产调用点在 tick 起点且 awaitCompletion=false，
    // inFlight 任务由 scheduler.inFlight 防重，员工重占用由 claim 原子性保证
  }

  private recordOf(id: string): EmployeeRecord | undefined {
    return this.records.find((r) => r.id === id);
  }

  /** 同步能力谓词：无 plan 不限；plan 各项 kind（缺省 dev）都必须在员工绑定内（空=全部可用） */
  canDispatch(task: TaskRecord, employee: EmployeeProfile): boolean {
    const caps = this.capabilitiesOf(employee.id);
    if (caps === undefined) return true;   // 不在档案（外部注入 profile，如测试）→ 不限制
    if (caps.length === 0) return true;    // 空 = 全部可用
    const kinds = kindsOf(task);
    return kinds.every((k) => caps.includes(k));
  }

  acquire(role?: string, extra?: (p: EmployeeProfile) => boolean): EmployeeProfile | null {
    const hit = this.records.find((r) => {
      if (!this.free.has(r.id)) return false;
      if (role && !r.skills.includes(role)) return false;
      if (extra) {
        const profile: EmployeeProfile = {
          id: r.id, name: r.name, role: r.role, skills: [...r.skills],
          ...(r.supervision ? { supervision: r.supervision } : {}),
        };
        if (!extra(profile)) return false;
      }
      return true;
    });
    if (!hit) return null;
    this.free.delete(hit.id);
    return {
      id: hit.id, name: hit.name, role: hit.role, skills: [...hit.skills],
      ...(hit.supervision ? { supervision: hit.supervision } : {}),
    };
  }

  release(id: string): void {
    if (this.records.some((r) => r.id === id)) this.free.add(id);
  }

  capabilitiesOf(id: string): string[] | undefined {
    return this.recordOf(id)?.capabilities;
  }

  /** 员工专属模型（装配 modelFor 用，2026-09-06 增补）：基于 refresh 快照同步访问；
   *  补齐 ModelSpec.name（员工专属命名，与 ModelSpec 同构）；未绑定返回 undefined */
  modelOf(id: string): ModelSpec | undefined {
    const m = this.recordOf(id)?.model;
    if (!m) return undefined;
    return {
      name: `${id}-${m.model}`, baseUrl: m.baseUrl, apiKey: m.apiKey, model: m.model,
      ...(m.api ? { api: m.api } : {}),
    };
  }
}
```

```ts
import type { AgentEvent, EmployeeOutcome } from '@ddw/runtime';
import type { EventStore, EventFilter, TaskStore, TaskRecord, TaskPackage } from '../stores/index.js';
import type { MessageRecord, MessageType } from './messages.js';

/** 消息出口（server 接线：FileMessageStore 落库 + 通知渠道推送；Task 6 前只落库） */
export interface MessageSink {
  onMessage(m: Omit<MessageRecord, 'id' | 'createdAt' | 'readAt'>): Promise<void>;
}

const msg = (type: MessageType, taskId: string, summary: string, employeeId?: string) => ({
  type, taskId, summary,
  title: type === 'review_required' ? `任务 ${taskId} 待人工放行`
    : type === 'task_failed' ? `任务 ${taskId} 执行失败`
    : `任务 ${taskId} 执行完成`,
  ...(employeeId ? { employeeId } : {}),
});

/** 事件侧装饰器：task_check awaiting=true → review_required；其余透传 */
export function wrapEventStoreForMessages(base: EventStore, sink: MessageSink): EventStore {
  return {
    append: async (e: AgentEvent) => {
      await base.append(e);
      if (e.type === 'task_check' && (e.payload as { awaiting?: boolean } | undefined)?.awaiting === true) {
        await sink.onMessage(msg('review_required', e.taskId, e.summary, e.employeeId));
      }
    },
    list: (f?: EventFilter) => base.list(f),
    ...(base.verifyIntegrity ? { verifyIntegrity: () => base.verifyIntegrity!() } : {}),
    ...(base.snapshotHead ? { snapshotHead: () => base.snapshotHead!() } : {}),
  };
}

/** 任务池侧装饰器：finish 落定 → task_done/task_failed；其余透传 */
export function wrapTaskStoreForMessages(base: TaskStore, sink: MessageSink): TaskStore {
  const finishMsg = (taskId: string, outcome: EmployeeOutcome, ok: boolean) =>
    msg(ok ? 'task_done' : 'task_failed', taskId, outcome.reply ?? '');
  return {
    add: (pkg: TaskPackage, opts?: { draft?: boolean }) => base.add(pkg, opts),
    get: (id: string) => base.get(id),
    list: () => base.list(),
    claim: (id: string, employeeId: string) => base.claim(id, employeeId),
    markRunning: (id: string) => base.markRunning(id),
    finish: async (taskId: string, outcome: EmployeeOutcome, ok: boolean, progress?, failedItemId?) => {
      const rec = await base.finish(taskId, outcome, ok, progress, failedItemId);
      await sink.onMessage(finishMsg(taskId, outcome, ok));
      return rec;
    },
    publish: (id: string) => base.publish(id),
    resumePlan: (id: string) => base.resumePlan(id),
  };
}
```

```ts
export function validateChannelDef(def: NotificationChannelDef): void {
  if (typeof def.id !== 'string' || !def.id.trim()) throw new Error('渠道 id 必须是非空字符串');
  if (!['dingtalk', 'wecom', 'webhook'].includes(def.type)) throw new Error('渠道 type 必须是 dingtalk | wecom | webhook');
  if (typeof def.name !== 'string' || !def.name.trim()) throw new Error('渠道 name 必须是非空字符串');
  if (typeof def.webhookUrl !== 'string' || !/^https?:\/\//.test(def.webhookUrl)) throw new Error('渠道 webhookUrl 必须是 http(s) 地址');
  if (typeof def.enabled !== 'boolean') throw new Error('渠道 enabled 必须是布尔值');
}

/** 钉钉加签：secret → `&timestamp=<ms>&sign=<urlencode(base64(hmac-sha256(secret)))>` */
function dingtalkSign(webhookUrl: string, secret: string): string {
  const ts = Date.now();
  const sign = createHmac('sha256', secret).update(`${ts}\n${secret}`).digest('base64');
  return `${webhookUrl}&timestamp=${ts}&sign=${encodeURIComponent(sign)}`;
}

/** 消息 → 渠道 payload（纯函数；consoleUrl 直达链接可空） */
export function buildChannelPayload(
  def: NotificationChannelDef, message: { type: string; title: string; summary: string; taskId: string }, consoleUrl?: string,
): { url: string; body: Record<string, unknown> } {
  const link = consoleUrl ? `${consoleUrl}/tasks/${message.taskId}` : '';
  const text = `${message.title}\n${message.summary}${link ? `\n直达: ${link}` : ''}`;
  if (def.type === 'dingtalk') {
    return {
      url: def.secret ? dingtalkSign(def.webhookUrl, def.secret) : def.webhookUrl,
      body: { msgtype: 'text', text: { content: text } },
    };
  }
  if (def.type === 'wecom') {
    return {
      url: def.webhookUrl,
      body: { msgtype: 'markdown', markdown: { content: `**${message.title}**\n${message.summary}${link ? `\n[直达](${link})` : ''}` } },
    };
  }
  return { url: def.webhookUrl, body: { ...message, ...(consoleUrl ? { consoleUrl } : {}) } };
}

/** 推送单渠道：非 2xx 抛错（调用方记日志不阻断） */
export async function pushToChannel(def: NotificationChannelDef, message: Parameters<typeof buildChannelPayload>[1], consoleUrl?: string, timeoutMs = 10_000): Promise<void> {
  const { url, body } = buildChannelPayload(def, message, consoleUrl);
  const res = await fetch(url, {
    method: 'POST', headers: { 'content-type': 'application/json' },
    body: JSON.stringify(body), signal: AbortSignal.timeout(timeoutMs),
  });
  if (!res.ok) throw new Error(`渠道 ${def.name} 推送失败: HTTP ${res.status}`);
}
```
